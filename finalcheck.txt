### 3c. _load_output (copy into test_outputs.py)
  def _load_output(path):
      """Load agent-produced output with symlink and hardlink protection."""
      assert os.path.exists(path), f"Output not found: {path}"
      assert not os.path.islink(path), f"Must not be a symlink: {path}"
      real_path = os.path.realpath(path)
      assert real_path.startswith("/app/"), f"Path escapes /app: {real_path}"
      out_stat = os.stat(path)
      assert out_stat.st_nlink == 1, f"Output has multiple hard links"
      with open(path, "r") as f:
          return json.load(f)

### 3d. approx_equal (copy into test_outputs.py)
  def approx_equal(a, b, rel_tol=1e-3):
      """Approximate equality — exact zero must match exactly."""
      if isinstance(a, (int, float)) and isinstance(b, (int, float)):
          if b == 0.0:
              return a == 0.0          # CRITICAL: p_value=0.0 must match exactly
          if abs(b) < 1e-10:
              return abs(a) < 1e-6
          return abs(a - b) / abs(b) < rel_tol
      return a == b

### 3e. input-derivation test (adapt for your task)
  def test_eval1_output_derived_from_input():
      """Prove output was computed by running the fixed pipeline on THIS specific input.
      Reads raw config and checks values that require running the fixed pipeline.
      """
      with open("/app/eval_config_1.json") as f:
          config = json.load(f)
      out = _load_output("/app/output_eval_1.json")

      # Structural tie to THIS config (not eval2 — must differ between evals)
      expected_ids = {s["session_id"] for s in config["sessions"]}
      assert {s["summary"]["session_id"] for s in out["sessions"]} == expected_ids
      assert out["session_count"] == len(config["sessions"])

      # Bug discriminator — value requires running FIXED pipeline, not readable from config
      # Threshold must DIFFER from eval2's threshold to block cross-config copying
      assert out["total_verification_failures"] == 0, \
          f"failures={out['total_verification_failures']} — bugs not fixed"

  def test_eval2_output_derived_from_input():

      with open("/app/eval_config_2.json") as f:

      assert out["total_application_records"] == total_recs

## PART 7 — BUG DESIGN (what works vs what fails)

### WHAT FAILS (7/7 self-play — never use):
  - Argument order swaps: compute_fn(a, b) instead of compute_fn(b, a) — agents read sig, instant fix
  - Formula errors: wrong constant, wrong exponent — models know all formulas
  - Docstring contradicts code — agents spot this immediately
  - Wrong function called when right function has a similar name nearby
  - Any bug a model can verify by "checking textbook"

### WHAT WORKS (0/5 CI):
  - CROSS-MODULE DATA-FLOW: wrong variable passed to correct function
    e.g., pipeline passes hello_hash to compute_application_traffic_secrets (needs handshake_hash)
    Comment justifies: "cached for efficiency" — invisible when hello_hash == handshake_hash
  - ORDERING/PLACEMENT: correct functions called in wrong sequence
    e.g., key derived from traffic_secret instead of confirmed_secret (after confirmation step)
    Comment justifies: "negotiation independence" — invisible when confirmation_required=False
  - EPOCH/CONDITION-AWARE: standard path used for all cases, special path never called
    e.g., build_nonce_standard always called even for epoch > 0 (needs build_nonce_extended)
    Comment justifies: "RFC 8446 §5.3 suffices" — invisible when initial_epoch=0

### MASKING RULES (visible data must make buggy == fixed EXACTLY):
  - hello_hash == handshake_hash → stale cache produces correct result
  - confirmation_required = False → confirmed_secret == traffic_secret
  - initial_epoch = 0 → build_nonce_standard == build_nonce_extended
  - Single session → no stale cache across sessions
  - n=1 batch → sum/n == sum (normalize-by-count bug invisible)

### DECOY RULES (correct code that looks wrong):
  - Must have impact ≥30% if removed (so agents who touch it get wrong output)
  - Protect explicitly in instruction.md: "preserve X which correctly does Y"
  - Examples: double-SHA fingerprint, session-bound domain separation, 4+8 byte nonce padding
  - Agents WILL modify protected decoys — that's the compound trap

### SELF-PLAY RESULT → CI PREDICTION:
  - 0/7 self-play → likely 0/5 CI (strong)
  - 1-2/7 self-play → likely 0-2/5 CI (acceptable)
  - 7/7 self-play → CI will likely pass too (too easy — redesign)
  - Exception: over-engineering trap (agents find bugs but also break decoys) can get
    7/7 self-play identified but still 0/5 CI because fixing + breaking decoy = fail

## PART 10 — OLD vs NEW TEST.SH PATTERN

Older accepted tasks (dynamo-37d03aa, dynamo-c75e93b) use this pattern:
  test.sh copies hidden eval configs from /tests/ to /app/, re-runs pipeline, then pytest
  eval configs live in tests/ (hidden from agent)

Newer QC_GATE pattern (dynamo-6b6e49d and later) uses this pattern:
  test.sh ONLY runs pytest — no pipeline calls
  eval configs live in environment/data/ (agent-visible, COPYed into image)
  Agent is instructed in instruction.md to run the pipeline on eval configs itself

Both patterns work on CI. The newer pattern is simpler and avoids AVA complaints
about "verifier re-runs agent code". Prefer the new pattern for all future tasks.
