## EXACT _load_output TEMPLATE

```python
def _load_output(path):
    """Load agent-produced output with symlink and hardlink protection."""
    assert os.path.exists(path), f"Output not found: {path}"
    assert not os.path.islink(path), f"Must not be a symlink: {path}"
    real_path = os.path.realpath(path)
    assert real_path.startswith("/app/"), f"Path escapes /app: {real_path}"
    # Hardlink guard — catches hardlink bypass attack
    out_stat = os.stat(path)
    assert out_stat.st_nlink == 1, f"Output {path} has multiple hard links"
    for fixture_dir in ["/tests"]:
        if os.path.isdir(fixture_dir):
            for fname in os.listdir(fixture_dir):
                fpath = os.path.join(fixture_dir, fname)
                try:
                    fix_stat = os.stat(fpath)
                    assert not (out_stat.st_ino == fix_stat.st_ino and
                                out_stat.st_dev == fix_stat.st_dev)
                except OSError:
                    pass
    with open(path, "r") as f:
        return json.load(f)
```

## EXACT approx_equal TEMPLATE (with b==0.0 fix)

```python
def approx_equal(a, b, rel_tol=1e-3):
    """Check approximate equality with relative tolerance."""
    if isinstance(a, (int, float)) and isinstance(b, (int, float)):
        if b == 0.0:              # CRITICAL: p_value=0.0 must match exactly
            return a == 0.0
        if abs(b) < 1e-10:
            return abs(a) < 1e-6
        return abs(a - b) / abs(b) < rel_tol
    return a == b
```

def _load_output(path):
    """Load an output file with symlink protection."""
    assert os.path.exists(path), f"Output not found: {path}"
    assert not os.path.islink(path), f"Output must not be a symlink: {path}"
    real_path = os.path.realpath(path)
    assert real_path.startswith("/app/"), f"Output escapes /app: {real_path}"
    with open(path, "r") as f:
        return json.load(f)
